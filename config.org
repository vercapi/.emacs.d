    "Copy the current buffer file name and path to the clipboard."
    (interactive)
    (let ((filename (if (equal major-mode 'dired-mode)
                        default-directory
                      (buffer-file-name))))
      (when filename
        (kill-new filename)
        (message "Copied buffer file name '%s' to the clipboard." filename))))

                        (string-match "[^/]+/*$" default-directory)

        (kill-new filename)

  (use-package bm
    :ensure t
    :demand t
    :init
    (setq bm-restore-repository-on-load t) ;; restore on load (even before you require bm)
    :config
    (setq bm-cycle-all-buffers nil  ;; Don't llow cross-buffer 'next'

    ;; Loading the repository from file when on start up.

    ;; Restoring bookmarks

*** Homing

Moving smater to the beginning of a line
    
[[https://emacsredux.com/blog/2013/05/22/smarter-navigation-to-the-beginning-of-a-line/][src]]

This is a short example
text|
# pressing C-a once
|text
# pressing C-a again
|   text
# pressing C-a again
|text

#+begin_src emacs-lisp
  (defun smarter-move-beginning-of-line (arg)
    "Move point back to indentation of beginning of line.

  Move point to the first non-whitespace character on this line.
  If point is already there, move to the beginning of the line.
  Effectively toggle between the first non-whitespace character and
  the beginning of the line.

  If ARG is not nil or 1, move forward ARG - 1 lines first.  If
  point reaches the beginning or end of the buffer, stop there."
    (interactive "^p")
    (setq arg (or arg 1))

    ;; Move lines first
    (when (/= arg 1)
      (let ((line-move-visual nil))
        (forward-line (1- arg))))

    (let ((orig-point (point)))
      (back-to-indentation)
      (when (= orig-point (point))
        (move-beginning-of-line 1))))

  ;; remap C-a to `smarter-move-beginning-of-line'
  (global-set-key [remap move-beginning-of-line]
                  'smarter-move-beginning-of-line)
#+end_src

** Spell Checking

*** Main setup

Set flyspell on all buffers automatically detect prog mode
#+begin_src emacs-lisp
  (defun flyspell-on-for-buffer-type ()
    "Enable Flyspell appropriately for the major mode of the current buffer.  Uses `flyspell-prog-mode' for modes derived from `prog-mode', so only strings and comments get checked.  All other buffers get `flyspell-mode' to check all text.  If flyspell is already enabled, does nothing."
    (interactive)
    (if (not (bound-and-true-p flyspell-mode)) ; if not already on
        (progn
          (if (or (string-match-p "helm" (if (listp mode-name)
                                             (car mode-name)
                                           mode-name)) (memq major-mode '(org-agenda-mode)))
              (message (format "No flyspell on %s" major-mode))
            ;; else
            (if (or (eq major-mode 'magit-status-mode) (derived-mode-p 'prog-mode))
                (progn
                  (message "Flyspell on (code)")
                  (flyspell-prog-mode))
              ;; else
              (progn
                (message "Flyspell on (text)")
                (flyspell-mode 1))))
          )))

  (defun flyspell-toggle ()
    "Turn Flyspell on if it is off, or off if it is on.  When turning on, it uses `flyspell-on-for-buffer-type' so code-vs-text is handled appropriately."
    (interactive)
    (if (bound-and-true-p flyspell-mode)
        (progn ; flyspell is on, turn it off
          (message "Flyspell off")
          (flyspell-mode -1))
      ;; else - flyspell is off, turn it on
      (flyspell-on-for-buffer-type)))
#+end_src

    (defun multi-git-exec (rootdir action)
      "Run the ACTION function for every git repo in ROOTDIR."
      (progn
        (with-current-buffer (get-buffer-create "*Multi git log*")
          (multi-git-mode)
          (let ((inhibit-read-only t))
            (erase-buffer)
            (insert (concat "Root directory: " rootdir))
            (newline 2)
            (mapc (lambda (dir)
                    (let ((default-directory (concat rootdir dir)))
                      (progn
                        (let ((items (funcall action)))
                          (if items
                              (progn
                                (insert (concat "* " dir " " (car items) " *"))
                                (newline 2)
                                (mapc (lambda (item)
                                        (insert (format "  %s"
                                                        (replace-regexp-in-string "\n$" "" item)))
                                        (newline))
                                      (cdr items))
                                (newline 2)))))))
                  (multi-git-dirs rootdir))))))

** Org workspace Magit

Two dedicated commands to open Magit for the Org workspace's two repos. The
workspace is versioned by two repos whose git-dirs live outside the work tree
and that *share the same work tree*: the private notes repo (=~/org-vcs/notes.git=)
and the shareable config repo (=~/org-vcs/config.git=).

The shared work tree is the crux. =git rev-parse --show-toplevel= returns the
same directory for both repos, so Magit keys a single status buffer on that
toplevel — that's fine here: only one repo is "live" at a time, and switching
repos just re-points which git-dir Magit uses.

The mechanism is *directory-scoped advice* on =magit-process-environment=. In
this Magit that is a function (no args) returning the list of env strings, and
it is called for *every* git subprocess in *every* Magit buffer (status, log,
diff, show, revert). We advise its return value: whenever =default-directory= is
inside the Org work tree, we append =GIT_DIR= / =GIT_WORK_TREE= pointing at the
currently-active repo (=pve/org-active-git-dir=). Because this keys off global
state plus =default-directory= (which every child buffer shares) rather than a
buffer-local =process-environment=, git resolution works uniformly across the
status buffer *and* all the fresh log/diff/show/revert buffers Magit spawns —
a buffer-local-env approach breaks exactly those child buffers with "fatal: not
a git repository". Outside the work tree the advice is a no-op, so normal repos
(this Emacs config, etc.) are unaffected.

=pve/org-magit= and =pve/org-config-magit= are thin interactive wrappers around
the shared =pve/org--magit-open=: each sets the active git-dir and opens (or
refreshes) the single status buffer. Re-running the other command switches which
git-dir all git calls in the work tree resolve against.
#+BEGIN_SRC emacs-lisp
  (defvar pve/org-worktree "/mnt/c/Users/vercapi/OneDrive - Cronos/Documenten/org"
    "Work-tree shared by the Org workspace's two external-git-dir repos.")

  (defvar pve/org-active-git-dir nil
    "Absolute git-dir Magit should use for git calls inside `pve/org-worktree'.
  Only one of the two repos can be live at a time (they share one work-tree).")

  (defun pve/org--magit-inject-env (env)
    "Append GIT_DIR/GIT_WORK_TREE to ENV for git calls inside the Org work-tree.
  Advice on `magit-process-environment' (a no-arg function returning the env
  list for every git subprocess in every Magit buffer).  Keys off global
  `pve/org-active-git-dir' plus `default-directory' — shared by all the fresh
  log/diff/show/revert buffers Magit spawns — so git resolves to the active repo
  everywhere, not just the status buffer.  A no-op outside `pve/org-worktree'."
    (if (and pve/org-active-git-dir
             default-directory
             (file-in-directory-p (expand-file-name default-directory)
                                  (expand-file-name pve/org-worktree)))
        (append (list (concat "GIT_DIR=" pve/org-active-git-dir)
                      (concat "GIT_WORK_TREE=" (expand-file-name pve/org-worktree)))
                env)
      env))

  (advice-add 'magit-process-environment :filter-return #'pve/org--magit-inject-env)

  (defun pve/org--magit-open (repo-name)
    "Open Magit on the Org work-tree for REPO-NAME's git-dir (\"notes\"/\"config\").
  Sets the active git-dir and opens (or refreshes) the shared status buffer;
  all git calls in the work-tree then resolve against that repo via the advice
  on `magit-process-environment' (see `pve/org--magit-inject-env')."
    (require 'magit)
    (setq pve/org-active-git-dir
          (expand-file-name (format "org-vcs/%s.git" repo-name) "~"))
    (magit-status-setup-buffer pve/org-worktree))

  (defun pve/org-magit ()
    "Open Magit for the private Org notes repo (browse history, revert)."
    (interactive)
    (pve/org--magit-open "notes"))

  (defun pve/org-config-magit ()
    "Open Magit for the shareable Org config repo (CLAUDE.md + latex/)."
    (interactive)
    (pve/org--magit-open "config"))

    "Creates a source for helm using lastpass command line"
    (mapcar

*** File setup

More or less loosly based on: [[http://howardism.org/Technical/Emacs/getting-more-boxes-done.html][Getting boxes done]]

The flow works as follows

1. everything goes into inbox
2. Go through inbox

     1. *Delete* if no longer relevant
     2. Create an *idea* that neads further work
     3. *References* for information that can be used later on
   - Actionable

     2. create a *tasklist* if small (Can grow into projects)
     3. Create a *project* for a big thing

** LaTeX export

Register the CraftCode brand document class (=craftcode.cls=, found via kpathsea
through a =~/texmf= symlink) with the LaTeX exporter. =[NO-DEFAULT-PACKAGES]= is
deliberate: the class loads everything itself (fontspec-based, incompatible with
the inputenc/fontenc defaults). =[PACKAGES]= and =[EXTRA]= stay so
=#+LATEX_HEADER= lines still work. The class styles source code via listings,
defines =\subtitle=, and its Lexend brand font requires LuaLaTeX.

Note the subtitle format: with =org-latex-subtitle-separate= set, ox-latex emits
the *formatted* subtitle string raw into the preamble right after =\title{}= —
so the format itself must wrap the text in the class's =\subtitle= macro. With
the default format the subtitle text lands unwrapped in the preamble and breaks
the title block.

#+BEGIN_SRC emacs-lisp
  (with-eval-after-load 'ox-latex
    (add-to-list 'org-latex-classes
                 '("craftcode"
                   "\\documentclass{craftcode}\n[NO-DEFAULT-PACKAGES]\n[PACKAGES]\n[EXTRA]"
                   ("\\section{%s}" . "\\section*{%s}")
                   ("\\subsection{%s}" . "\\subsection*{%s}")
                   ("\\subsubsection{%s}" . "\\subsubsection*{%s}")
                   ("\\paragraph{%s}" . "\\paragraph*{%s}")
                   ("\\subparagraph{%s}" . "\\subparagraph*{%s}")))
    (setq org-latex-src-block-backend 'listings)   ; class styles code via listings
    (setq org-latex-subtitle-separate t)           ; class defines \subtitle
    (setq org-latex-subtitle-format "\\subtitle{%s}") ; see prose note above
    (setq org-latex-tables-booktabs t)             ; class loads booktabs
    (setq org-latex-compiler "lualatex")           ; Lexend brand font needs LuaLaTeX
    (setq org-latex-pdf-process
          '("latexmk -f -pdf -pdflatex=lualatex -interaction=nonstopmode -output-directory=%o %f")))
#+END_SRC

* AI Helpers

Elisp helper functions created for (and by) Claude AI-assistant workflows
that operate on the org workspace through the Emacs MCP server. Claude calls
these via the MCP =eval-elisp= tool. They are added and modified via agents
spawned from that workspace — not scattered through the topical config
sections above, so it stays clear which code exists purely for the AI
workflows. Each helper here must also be documented in the org workspace's
=CLAUDE.md= (the "Custom commands" table in the Emacs MCP section) so Claude
sessions know to use it. New helpers follow the surrounding naming style
(=pve/= / =pve-= prefix); no special AI-specific prefix is required.

** pve/org-set-state

A programmatic ~org-todo~ call (and the MCP node-update tool) skips the
deferred state-change log note, because Org defers the log-note
insertion to the command loop, which never runs in a headless call.
This helper sets the state and then flushes the pending note
explicitly so every status change lands in the =:LOGBOOK:= drawer.
The =save-buffer= persists the change, since these files are also
edited outside Emacs.
#+BEGIN_SRC emacs-lisp
  (defun pve/org-set-state (id state)
    "Set the TODO keyword of the Org node with ID to STATE, logging the
  transition into its :LOGBOOK: drawer.  STATE is a keyword string such
  as \"REVIEW\"; pass an empty string to clear the state.  Returns STATE.

  Needed because a programmatic `org-todo' (and the MCP node-update tool)
  skip the deferred state-change log note; this flushes it explicitly.

  REVIEWED is reserved for the user and must never be set through this
  helper; it is only ever set interactively via `C-c C-t'."
    (when (member state '("REVIEWED"))
      (user-error "REVIEWED is reserved for the user; it must not be set programmatically"))
    (save-window-excursion
      (org-id-goto id)
      (org-todo state)
      (when (and (markerp org-log-note-marker) (marker-buffer org-log-note-marker))
        (org-add-log-note))
      (save-buffer))
    state)
#+END_SRC

#+BEGIN_SRC emacs-lisp
  (defun pve/org-questions-meta-return ()
    "In a :QUESTIONS: drawer, insert an answer line (-> ) instead of a new item.
  Intended as an `org-metareturn-hook' function: returns t to skip the
  default `org-meta-return' behaviour when point is inside a :QUESTIONS:
  drawer and on a list-item line, nil otherwise."
    (when (and (derived-mode-p 'org-mode)
               ;; Is point on a list-item line?
               (save-excursion
                 (beginning-of-line)
                 (looking-at "[ \t]*- "))
               ;; Is point inside a :QUESTIONS: drawer?
               (save-excursion
                 (let ((drawer-start
                        (re-search-backward "^[ \t]*:QUESTIONS:" nil t))
                       (drawer-end
                        (progn
                          (goto-char (point))
                          (re-search-forward "^[ \t]*:END:" nil t))))
                   (and drawer-start drawer-end
                        ;; Make sure the :QUESTIONS: we found really closes
                        ;; with the :END: that comes *after* point.
                        (progn
                          (goto-char drawer-start)
                          (re-search-forward "^[ \t]*:END:" nil t))

                        (let ((pos (point-marker)))

** pve/add-ailog-entry

Append a timestamped note to the =:AILOG:= drawer of the Org node with the
given ID.  If the drawer already exists, the entry is inserted on a new line
just before the closing =:END:=.  If no =:AILOG:= drawer exists yet, one is
created immediately after the node's meta-data block.

The =re-search-forward= for the drawer open-line uses =^:AILOG:= (no =$=
anchor) so it matches lines that have trailing whitespace or tags after the
drawer name — a strict =^:AILOG:$= would miss those and fall through to the
else branch, creating a duplicate drawer.
#+BEGIN_SRC emacs-lisp
  (defun pve/add-ailog-entry (id note)
    "Append NOTE to the :AILOG: drawer of the Org node identified by ID.
  If the drawer exists it is extended; otherwise a new one is created after
  the node's meta-data block.  The buffer is saved after the insertion."
    (save-window-excursion
      (org-id-goto id)
      (org-back-to-heading t)
      (let* ((beg (point))
             (end (save-excursion (org-end-of-subtree t t) (point))))
        (goto-char beg)
        (if (re-search-forward "^:AILOG:" end t)
            (progn
              (re-search-forward "^:END:$" end t)
              (beginning-of-line)
              (insert (format "[%s] %s\n"
                              (format-time-string "%Y-%m-%d")
                              note)))
          (goto-char beg)
          (org-end-of-meta-data t)
          (insert (format ":AILOG:\n[%s] %s\n:END:\n"
                          (format-time-string "%Y-%m-%d")
                          note)))
        (save-buffer))))
#+END_SRC

** pve/org-aiinstructions

Ancestor-aware reader for =:AIINSTRUCTIONS:= drawers. Given a node ID it
collects every applicable drawer in scope — the file-level drawer (before the
first heading), each ancestor heading's drawer, and the node's own drawer —
ordered broadest to closest so the caller can apply closest-wins. The result
is a single labelled string (=[file]=, =[ancestor: TITLE]=, =[node: TITLE]=),
or nil when no such drawer exists in any scope. It is a pure reader and never
saves the buffer.

The two private helpers do the region scanning. Each restricts its search to a
region bounded by the next heading, so a *descendant's* drawer is never picked
up for an ancestor. The opener is matched with =^[ \t]*:AIINSTRUCTIONS:= (no
=$= anchor, per the drawer-regexp gotcha — a trailing tag or whitespace on the
line would otherwise defeat the match).
#+BEGIN_SRC emacs-lisp
  (defun pve/org--aiinstructions-in-region (beg end)
    "Return body of the first :AIINSTRUCTIONS: drawer within [BEG,END), or nil.
  The opener is matched with ^[ \\t]*:AIINSTRUCTIONS: (no $ anchor, per the
  drawer-regexp gotcha).  Body is the text strictly between the opener line and
  its matching ^[ \\t]*:END:, markers excluded and trimmed."
    (save-excursion
      (goto-char beg)
      (when (re-search-forward "^[ \t]*:AIINSTRUCTIONS:" end t)
        (forward-line 1)
        (let ((body-start (point))
              (body-end (if (re-search-forward "^[ \t]*:END:" end t)
                            (line-beginning-position) end)))
          (string-trim (buffer-substring-no-properties body-start body-end))))))

  (defun pve/org--aiinstructions-in-heading-region ()
    "Return the :AIINSTRUCTIONS: body for the heading at point, or nil.
  Searches only from the heading to the next heading of any level, so a
  descendant's drawer is never picked up."
    (save-excursion
      (org-back-to-heading t)
      (let ((beg (line-beginning-position))
            (end (save-excursion
                   (forward-line 1)
                   (if (re-search-forward "^\\*+ " nil t)
                       (line-beginning-position) (point-max)))))
        (pve/org--aiinstructions-in-region beg end))))

  (defun pve/org-aiinstructions (id)
    "Return all applicable :AIINSTRUCTIONS: drawer bodies in scope for node ID.
  Collects the file-level drawer (before the first heading), every ancestor
  heading's drawer, and the node's own drawer, ordered broadest to closest so
  the caller applies closest-wins.  Returns a single labeled string (scopes
  labeled [file], [ancestor: TITLE], [node: TITLE]), or nil when no
  :AIINSTRUCTIONS: drawer exists in any scope.  Pure reader: does not save."
    (save-window-excursion
      (org-id-goto id)
      (org-back-to-heading t)
      (let ((scopes '()))
        (let ((body (pve/org--aiinstructions-in-heading-region)))
          (when body
            (push (cons (format "[node: %s]" (org-get-heading t t t t)) body) scopes)))
        (save-excursion
          (while (org-up-heading-safe)
            (let ((body (pve/org--aiinstructions-in-heading-region)))
              (when body
                (push (cons (format "[ancestor: %s]" (org-get-heading t t t t)) body) scopes)))))
        (save-excursion
          (goto-char (point-min))
          (let* ((first-heading (save-excursion
                                  (if (re-search-forward "^\\*+ " nil t)
                                      (line-beginning-position) (point-max))))
                 (body (pve/org--aiinstructions-in-region (point-min) first-heading)))
            (when body
              (push (cons "[file]" body) scopes))))
        (when scopes
          (concat (mapconcat (lambda (s) (format "%s\n%s" (car s) (cdr s))) scopes "\n\n")
                  "\n\n(closest scope wins: the last block above overrides the earlier ones.)")))))
#+END_SRC
